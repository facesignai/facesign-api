## Basic Identity Verification

### Simple Liveness Check

The most basic flow for verifying a person is physically present and not using a photo or video.

<SimpleCodeBlock
  title="Basic Liveness Detection"
  language="typescript"
  code={`import { Client, FSNodeType, FSTwoFactorChannel } from '@facesignai/api'

const client = new Client({
  auth: process.env.FACESIGN_API_KEY
})

const { session, clientSecret } = await client.session.create({
  clientReferenceId: 'user-123',
  flow: {
    nodes: [
      {
        id: 'start',
        type: FSNodeType.START,
        outcome: 'liveness'
      },
      {
        id: 'liveness',
        type: FSNodeType.LIVENESS_DETECTION,
        outcomes: {
          livenessDetected: 'end',
          deepfakeDetected: 'end',
          noFace: 'end'
        }
      },
      {
        id: 'end',
        type: FSNodeType.END
      }
    ],
    edges: [
      { id: 'e1', source: 'start', target: 'liveness' },
      { id: 'e2', source: 'liveness', target: 'end' }
    ]
  }
})`}
/>

### Multi-Factor Authentication Flow

Combine email and SMS verification with biometric checks.

<SimpleCodeBlock
  title="Email + SMS + Biometric"
  language="typescript"
  code={`const { session, clientSecret } = await client.session.create({
  clientReferenceId: 'user-123',

    userEmail: 'user@example.com',

    nodes: [
      {
        id: 'start',
        type: FSNodeType.START,
        outcome: 'email'

        outcomes: {
          emailEntered: 'email_otp'

      {

## Conditional Flow Patterns

### Risk-Based Verification

Adjust verification requirements based on risk assessment.

<SimpleCodeBlock
  title="Dynamic Risk-Based Flow"
  language="typescript"
  code={`// Determine risk level based on your business logic
const riskLevel = await assessUserRisk(userId)  // 'low', 'medium', 'high'

// Build flow based on risk
const buildRiskBasedFlow = (risk: string) => {
  const baseNodes = [
    {
      id: 'start',
      type: FSNodeType.START,
      outcome: risk === 'low' ? 'liveness' : 'doc'
    },
    {
      id: 'end',
      type: FSNodeType.END
    }
  ]

  const edges = []

  if (risk === 'low') {
    // Low risk: Just liveness
    baseNodes.push({
      id: 'liveness',
      type: FSNodeType.LIVENESS_DETECTION,
      outcomes: {
        livenessDetected: 'end',
        deepfakeDetected: 'end',
        noFace: 'end'
      }
    })
    edges.push(
      { id: 'e1', source: 'start', target: 'liveness' },
      { id: 'e2', source: 'liveness', target: 'end' }
    )
  }

  if (risk === 'medium') {
    // Medium risk: Doc + Liveness
    baseNodes.push(
      {
        id: 'doc',
        type: FSNodeType.DOCUMENT_SCAN,
        outcomes: {
          verified: 'liveness',
          fraudDetected: 'end',
          manualReview: 'end',
          invalidDocument: 'end'
        }
      },
      {
        id: 'liveness',
        type: FSNodeType.LIVENESS_DETECTION,
        outcomes: {
          livenessDetected: 'end',
          deepfakeDetected: 'end',
          noFace: 'end'
        }
      }
    )
    edges.push(
      { id: 'e1', source: 'start', target: 'doc' },
      { id: 'e2', source: 'doc', target: 'liveness' },
      { id: 'e3', source: 'liveness', target: 'end' }
    )
  }

  if (risk === 'high') {
    // High risk: Full KYC + 2FA
    baseNodes.push(
      {
        id: 'doc',
        type: FSNodeType.DOCUMENT_SCAN,
        enhancedFraudDetection: true,
        outcomes: {
          verified: 'face',
          fraudDetected: 'end',
          manualReview: 'end',
          invalidDocument: 'end'
        }
      },
      {
        id: 'face',
        type: FSNodeType.FACE_SCAN,
        usePreviousImage: 'doc',
        outcomes: {
          matched: 'liveness',
          notMatched: 'end',
          noFace: 'end'
        }
      },
      {
        id: 'liveness',
        type: FSNodeType.LIVENESS_DETECTION,
        enhancedDetection: true,
        outcomes: {
          livenessDetected: 'sms',
          deepfakeDetected: 'end',
          noFace: 'end'
        }
      },
      {
        id: 'sms',
        type: FSNodeType.TWO_FACTOR,
        channels: [FSTwoFactorChannel.SMS],
        contactSource: 'session_data',
        staticPhone: userPhone,
        outcomes: {
          verified: 'end',
          failed_unverified: 'end',
          delivery_failed: 'end',
          cancelled: 'end',
          error: 'end'
        }
      }
    )
    edges.push(
      { id: 'e1', source: 'start', target: 'doc' },
      { id: 'e2', source: 'doc', target: 'face' },
      { id: 'e3', source: 'face', target: 'liveness' },
      { id: 'e4', source: 'liveness', target: 'sms' },
      { id: 'e5', source: 'sms', target: 'end' }
    )
  }

  return { nodes: baseNodes, edges }
}

// Create session with risk-appropriate flow
const { session, clientSecret } = await client.session.create({
  clientReferenceId: userId,
  metadata: {
    riskLevel: riskLevel,
    assessmentScore: riskScore
  },
  flow: buildRiskBasedFlow(riskLevel),
  webhookUrl: 'https://api.example.com/webhooks/verification'
})`}
