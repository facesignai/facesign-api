## Configuring Your Application

### Automatic Environment Detection

The SDK automatically routes to the correct environment based on your API key:

<SimpleCodeBlock
  title="Automatic Environment Routing"
  defaultLanguage="javascript"
  languageSwitcher="tabs"
  codeExamples={{
    javascript: `import { Client } from '@facesignai/api'

// SDK automatically detects environment from key prefix
const client = new Client({
  auth: process.env.FACESIGN_API_KEY
})

// sk_test_... → https://api.dev.facesign.ai
// sk_live_... → https://api.facesign.ai`,
    python: `import facesignai

# SDK automatically detects environment from key prefix
client = facesignai.Client(
    auth=os.environ.get('FACESIGN_API_KEY')
)

# sk_test_... → https://api.dev.facesign.ai
# sk_live_... → https://api.facesign.ai`,
    go: `import facesign "github.com/facesignai/go-sdk"

// SDK automatically detects environment from key prefix
client := facesign.NewClient(
    facesign.WithAPIKey(os.Getenv("FACESIGN_API_KEY"))
)

// sk_test_... → https://api.dev.facesign.ai
// sk_live_... → https://api.facesign.ai`
  }}
/>

### Environment Switching Code

<SimpleCodeBlock
  title="Environment-Aware Configuration"
  language="javascript"
  code={`class FaceSignService {
  constructor() {
    this.isDevelopment = process.env.NODE_ENV !== 'production'

    this.client = new Client({
      auth: this.isDevelopment
        ? process.env.FACESIGN_TEST_KEY
        : process.env.FACESIGN_LIVE_KEY,
      serverUrl: this.isDevelopment
        ? 'https://api.dev.facesign.ai'
        : 'https://api.facesign.ai'
    })

    // Different webhook URLs
    this.webhookUrl = this.isDevelopment
      ? process.env.DEV_WEBHOOK_URL
      : process.env.PROD_WEBHOOK_URL

    // Different logging levels
    this.logLevel = this.isDevelopment ? 'debug' : 'error'
  }

  async createSession(params) {
    try {
      const response = await this.client.session.create({
        ...params,
        webhookUrl: this.webhookUrl,
        metadata: {
          ...params.metadata,
          environment: this.isDevelopment ? 'dev' : 'prod'
        }
      })

      if (this.isDevelopment) {
        console.debug('Session created:', response)
      }

      return response
    } catch (error) {
      if (this.isDevelopment) {
        console.error('Detailed error:', error)
      }
      throw error
    }
  }
}`}
/>

## Monitoring and Debugging

### Development Debugging

<SimpleCodeBlock
  title="Enhanced Development Logging"
  language="javascript"
  code={`if (process.env.NODE_ENV === 'development') {
  // Log all API requests
  client.on('request', (config) => {
    console.log('API Request:', {
      method: config.method,
      url: config.url,
      data: config.data
    })
  })

  // Log all responses
  client.on('response', (response) => {
    console.log('API Response:', {
      status: response.status,
      data: response.data
    })
  })

  // Log all errors
  client.on('error', (error) => {
    console.error('API Error:', error)
  })
}`}
/>

### Production Monitoring

<SimpleCodeBlock
  title="Production Metrics Tracking"
  language="javascript"
  code={`// Track key metrics in production
class MetricsCollector {
  trackSession(event, data) {
    if (process.env.NODE_ENV === 'production') {
      // Send to your analytics service
      analytics.track({
        event: \`facesign_\${event}\`,
        properties: {
          sessionId: data.sessionId,
          duration: data.duration,
          success: data.success,
          // Never log PII
        }
      })
    }
  }

  trackError(error) {
    if (process.env.NODE_ENV === 'production') {
      // Send to error tracking service
      sentry.captureException(error, {
        tags: {
          service: 'facesign',
          environment: 'production'
        }
      })
    }
  }
}`}
/>
