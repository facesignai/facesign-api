      <Property name="secret" type="string">Opaque token for accessing the hosted verification interface. Format: `cs` prefix followed by 32 hex characters.</Property>
      <Property name="createdAt" type="number">Unix timestamp (milliseconds) when the client secret was created.</Property>
      <Property name="expireAt" type="number">Unix timestamp (milliseconds) when the client secret expires (2 hours after creation).</Property>
      <Property name="url" type="string">Complete hosted verification URL with embedded client secret query parameter.</Property>
